python main.py --url http://localhost:8080

# Scan with authentication
python main.py --url http://localhost:8080 \
  --auth-url /login.php \
  --auth-user admin \

# Run validation tests

# Debug authentication
python debug_auth.py --url http://localhost:8080 \
